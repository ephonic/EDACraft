cmake_minimum_required(VERSION 3.20)
project(FTimer VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
option(TIME_ENGINE_BUILD_CLI "Build standalone ftimer" ON)
option(TIME_ENGINE_ENABLE_TBB "Enable optional TBB parallel backend" ON)
option(TIME_ENGINE_ENABLE_NATIVE_DB "Build experimental native .db tools" OFF)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
set(CMAKE_INSTALL_LIBDIR lib CACHE STRING "Library directory within the standalone installation")
include(GNUInstallDirs)
include(cmake/Core.cmake)
include(cmake/EngineSources.cmake)
add_library(time_engine ${TIME_ENGINE_SOURCES})
target_compile_features(time_engine PUBLIC cxx_std_17)
add_library(TimeEngine::time_engine ALIAS time_engine)
target_include_directories(time_engine PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(time_engine PUBLIC edacraft_core Threads::Threads ZLIB::ZLIB)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS "9.0")
    target_link_libraries(time_engine PUBLIC stdc++fs)
endif()
if(UNIX)
    target_compile_definitions(time_engine PRIVATE HAVE_MMAP)
endif()
set(FTIMER_WITH_TBB OFF)
if(TIME_ENGINE_ENABLE_TBB)
    find_package(TBB QUIET)
    if(TBB_FOUND)
        target_link_libraries(time_engine PUBLIC TBB::tbb)
        target_compile_definitions(time_engine PUBLIC TIME_ENGINE_HAS_TBB)
        set(FTIMER_WITH_TBB ON)
    endif()
endif()
if(TIME_ENGINE_BUILD_CLI)
    add_executable(time_engine_cli src/cli/main.cc)
    set_target_properties(time_engine_cli PROPERTIES OUTPUT_NAME ftimer)
    target_link_libraries(time_engine_cli PRIVATE time_engine)
    install(TARGETS time_engine_cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
add_executable(lib2ftlib src/tools/lib2ftlib.cc)
target_include_directories(lib2ftlib PRIVATE src)
target_link_libraries(lib2ftlib PRIVATE time_engine)
install(TARGETS lib2ftlib RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(TIME_ENGINE_ENABLE_NATIVE_DB)
    add_library(time_engine_db_experimental STATIC src/liberty/db_api.cc
        src/liberty/db_native_model.cc src/liberty/liberty_text_writer.cc)
    target_link_libraries(time_engine_db_experimental PUBLIC time_engine)
    target_include_directories(time_engine_db_experimental PRIVATE src)
    add_executable(db_native_inspect src/tools/db_native_inspect.cc)
    target_link_libraries(db_native_inspect PRIVATE time_engine_db_experimental)
endif()
install(TARGETS time_engine edacraft_core EXPORT TimeEngineTargets
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES python/ftimer.py python/ftimer_sdc.py DESTINATION python)
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/ftimer/examples)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/ftimer)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/ftimer/docs)
install(FILES python/README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/ftimer/python)
install(FILES env.sh DESTINATION .)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/TimeEngineConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/TimeEngineConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TimeEngine)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/TimeEngineConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT TimeEngineTargets NAMESPACE TimeEngine::
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TimeEngine)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/TimeEngineConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/TimeEngineConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TimeEngine)
