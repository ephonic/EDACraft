cmake_minimum_required(VERSION 3.19)
project(dsim VERSION 0.13.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
  add_compile_options(-fp-model=strict)
endif()
include(CTest)
add_subdirectory(parsers)
add_library(dsim_source STATIC src/gate_source.cpp src/preprocessor.cpp src/time.cpp src/sdf.cpp src/timing_model.cpp)
target_include_directories(dsim_source PUBLIC src)
target_compile_options(dsim_source PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dsim_sdf_models src/sdf_models_main.cpp src/sdf_models.cpp src/liberty_sdf_conditions.cpp)
target_link_libraries(dsim_sdf_models PRIVATE dsim_source dsim_format_parsers)
target_compile_options(dsim_sdf_models PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dsim src/main.cpp src/rtl_compiler.cpp src/gate_compiler.cpp src/gate_partition.cpp src/gate_names.cpp src/sdf_binding.cpp src/sdf_delay.cpp src/sdf_annotation.cpp src/gate_timing.cpp)
target_link_libraries(dsim PRIVATE dsim_source)
target_link_libraries(dsim PRIVATE dsim_rtl_parser dsim_format_parsers)
target_include_directories(dsim PRIVATE src)
set(model_cxx_default "${CMAKE_CXX_COMPILER}")
if(APPLE AND CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang")
  # The SDK-aware system driver supplies libc++ headers; invoking Xcode's
  # underlying compiler directly can lose that SDK setup outside CMake.
  set(model_cxx_default "/usr/bin/clang++")
endif()
set(DSIM_MODEL_CXX "${model_cxx_default}" CACHE STRING "Default compiler for generated simulation models")
set(DSIM_MODEL_OPTIMIZATION "-O2" CACHE STRING "Default optimization for generated simulation models")
if(NOT DSIM_MODEL_OPTIMIZATION MATCHES "^-O[0123sgz]$")
  message(FATAL_ERROR "DSIM_MODEL_OPTIMIZATION must be an optimization level such as -O2 or -O3")
endif()
target_compile_definitions(dsim PRIVATE DSIM_DEFAULT_CXX="${DSIM_MODEL_CXX}" DSIM_DEFAULT_OPTIMIZATION="${DSIM_MODEL_OPTIMIZATION}")
set(DSIM_MODEL_CXX_FLAGS "" CACHE STRING "Additional flags for the default model compiler (semicolon-separated)")
set(DSIM_MODEL_FLAGS_CPP "")
foreach(model_flag IN LISTS DSIM_MODEL_CXX_FLAGS)
  string(REPLACE "\\" "\\\\" model_flag "${model_flag}")
  string(REPLACE "\"" "\\\"" model_flag "${model_flag}")
  string(REPLACE "\n" "\\n" model_flag "${model_flag}")
  string(REPLACE "\r" "\\r" model_flag "${model_flag}")
  string(APPEND DSIM_MODEL_FLAGS_CPP "\"${model_flag}\",")
endforeach()
configure_file(src/model_compiler_config.hpp.in model_compiler_config.hpp @ONLY)
target_include_directories(dsim PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set(DSIM_RUNTIME_DIR "${CMAKE_CURRENT_SOURCE_DIR}/include/dsim" CACHE PATH "Runtime header directory embedded in the compiler")
foreach(runtime_header IN ITEMS CONSTRAINT:constraint DPI:dpi RUNTIME:runtime WAVEFORM:waveform PATH:path PATH_DICTIONARY:path_dictionary WIDE:wide_value PARALLEL:parallel PARTITION:partition PARALLEL_WAVE:parallel_waveform)
  string(REPLACE ":" ";" runtime_pair "${runtime_header}")
  list(GET runtime_pair 0 runtime_macro)
  list(GET runtime_pair 1 runtime_file)
  target_compile_definitions(dsim PRIVATE DSIM_${runtime_macro}_PATH="${DSIM_RUNTIME_DIR}/${runtime_file}.hpp")
endforeach()
find_path(DSIM_ZSTD_INCLUDE zstd.h HINTS /opt/homebrew/include)
find_library(DSIM_ZSTD_LIBRARY NAMES zstd HINTS /opt/homebrew/lib)
if(DSIM_ZSTD_INCLUDE AND DSIM_ZSTD_LIBRARY)
  target_compile_definitions(dsim PRIVATE DSIM_ZSTD_INCLUDE="${DSIM_ZSTD_INCLUDE}" DSIM_ZSTD_LIBRARY="${DSIM_ZSTD_LIBRARY}")
endif()
find_path(DSIM_Z3_INCLUDE z3++.h HINTS /opt/homebrew/include)
find_library(DSIM_Z3_LIBRARY NAMES z3 HINTS /opt/homebrew/lib)
if(DSIM_Z3_INCLUDE AND DSIM_Z3_LIBRARY)
  target_compile_definitions(dsim PRIVATE DSIM_Z3_INCLUDE="${DSIM_Z3_INCLUDE}" DSIM_Z3_LIBRARY="${DSIM_Z3_LIBRARY}")
endif()
target_compile_options(dsim PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  include(cmake/Tests.cmake)
endif()
