cmake_minimum_required(VERSION 3.19)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(dsim_parsers LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    include(CTest)
endif()
option(DSIM_PARSERS_BUILD_RTL "Build the RTL parser (requires Tcl)" ON)
# Parsers maintained directly in dsim. RTL and gate netlists use separate code.
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)

add_library(dsim_gate_parser STATIC gate/gate_netlist_parser.cc gate/verilog_parser_adapter.cc)
add_library(dsim_spef_parser STATIC spef/spef_parser.cc)
add_library(dsim_liberty_parser STATIC liberty/liberty_parser.cc liberty/liberty_binary.cc liberty/liberty_compress.cc liberty/db_api.cc)
foreach(parser IN ITEMS dsim_gate_parser dsim_spef_parser dsim_liberty_parser)
    target_include_directories(${parser} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/common)
    target_link_libraries(${parser} PUBLIC Threads::Threads)
endforeach()
target_link_libraries(dsim_liberty_parser PUBLIC ZLIB::ZLIB)
add_library(dsim_format_parsers INTERFACE)
target_link_libraries(dsim_format_parsers INTERFACE dsim_gate_parser dsim_spef_parser dsim_liberty_parser)
add_executable(lib2ftlib tools/lib2ftlib.cc)
target_link_libraries(lib2ftlib PRIVATE dsim_liberty_parser)

if(DSIM_PARSERS_BUILD_RTL)
# Generated lexer/parser are checked in; normal builds do not need Flex/Bison.
find_package(TCL REQUIRED)
file(GLOB_RECURSE rtl_sources CONFIGURE_DEPENDS rtl/src/*.cpp)
add_library(dsim_rtl_parser STATIC ${rtl_sources})
set_target_properties(dsim_rtl_parser PROPERTIES CXX_STANDARD 11)
target_include_directories(dsim_rtl_parser PUBLIC
    rtl/src/err rtl/src/err/parser_err rtl/src/err/util_err rtl/src/mgr
    rtl/src/parser/verilog rtl/src/preproc/verilog rtl/src/tcl_engine
    rtl/src/token rtl/src/util/singleton ${TCL_INCLUDE_PATH})
target_link_libraries(dsim_rtl_parser PUBLIC ${TCL_LIBRARY} Threads::Threads)
add_executable(rtl_parser_demo rtl/vsyn.cpp)
set_target_properties(rtl_parser_demo PROPERTIES CXX_STANDARD 11)
target_link_libraries(rtl_parser_demo PRIVATE dsim_rtl_parser)
if(BUILD_TESTING)
    add_executable(test_verilog_semantic_memory rtl/test/semantic_memory.cpp)
    target_link_libraries(test_verilog_semantic_memory PRIVATE dsim_rtl_parser)
    if(MSVC)
        target_compile_options(test_verilog_semantic_memory PRIVATE /UNDEBUG)
    else()
        target_compile_options(test_verilog_semantic_memory PRIVATE -UNDEBUG)
    endif()
    add_test(NAME rtl_semantic_memory COMMAND test_verilog_semantic_memory)
endif()
endif()
if(BUILD_TESTING)
    add_executable(parser_bundle_smoke test/examples/parser_bundle_smoke.cc)
    target_link_libraries(parser_bundle_smoke PRIVATE dsim_format_parsers)
    add_test(NAME parser_bundle_smoke COMMAND parser_bundle_smoke "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
endif()
